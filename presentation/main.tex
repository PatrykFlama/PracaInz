\begin{frame}{Definicja problemu}

      Deterministyczny automat skończony to krotka:
      $$(Q, \Sigma, \delta, q_\lambda, \mathbb{F_A}, \mathbb{F_R})$$

      \begin{itemize}

        \item $Q$ - zbiór stanów

        \item $\delta: Q \times \Sigma \to Q$ - przejścia
        \item $q_\lambda$ - stan początkowy
        \item $\mathbb{F_A}$ - stany akceptujące
        \item $\mathbb{F_R}$ - stany odrzucające
      \end{itemize}
    \end{block}

\begin{frame}{Dlaczego ten problem jest ważny?}
  \begin{block}{Znany trudny problem}
    Problem \textit{DFA-consistency} (najmniejszy zgodny automat):
    \begin{itemize}
      \item \textbf{Wejście:} liczba $n$, zbiory $S^+$, $S^-$
      \item \textbf{Wyjście:} Czy istnieje DFA z $\leq n$ stanami akceptujący $S^+$ i odrzucający $S^-$?
      \item \textbf{Znany wynik:} $\mathbb{NP}$-zupełny
    \end{itemize}
  \end{block}
  
  \vspace{0.3cm}
  
  \begin{block}{Nasz problem}
    \begin{itemize}
      \item Generalnie co najmniej tak trudny co DFA-consistency
      \item Można pokazać redukcję z DFA-consistency
      \item Silnie motywuje teoretyczną analizę naszego problemu
    \end{itemize}
  \end{block}
\end{frame}

\section{Złożoność obliczeniowa}
\begin{frame}{NP-zupełność}
  \begin{block}{Cel: wykazać, że problem jest NP-zupełny}
    Trzeba pokazać dwie rzeczy:
    \begin{enumerate}
      \item Problem jest w $\mathbb{NP}$ (certyfikat istnieje)
      \item Problem jest $\mathbb{NP}$-trudny (nie ma algorytmu wielomianowego)
    \end{enumerate}
  \end{block}
  
  \vspace{0.1cm}
  
  \begin{block}{Problem jest w $\mathbb{NP}$}
    Dla danego rozwiązania (uzupełnienia automatu) weryfikacja zajmuje czas wielomianowy:
    \begin{itemize}
      \small
      \item Determinizm automatu: $O(|Q| \cdot |\Sigma|)$
      \item Sprawdzenie próbek: $O((|S^+| + |S^-|) \cdot M)$ gdzie $M$ to max długość
    \end{itemize}
  \end{block}
  
  \vspace{0.1cm}
  
  \begin{block}{Problem jest $\mathbb{NP}$-trudny}
    \textbf{Redukcja z DFA-consistency:}
    
    Dla instancji $(n, S^+, S^-)$ tworzymy częściowy automat z $n$ stanami, wszystkimi przejściami oraz wartościowaniami stanów niezdefiniowanymi. Jeśli można go naprawić, to rozwiązuje DFA-consistency.
  \end{block}
\end{frame}

\section{Złożoność parametryczna}
\begin{frame}{Złożoność parametryczna - motywacja}
  \begin{block}{Problem: nasz problem jest NP-zupełny}
    To oznacza, że nie istnieje znany wielomianowy algorytm dla ogólnego przypadku.
  \end{block}
  
  \vspace{0.2cm}
  
  \begin{block}{Obserwacja}
    Jeśli automat ma tylko \textbf{1 brakującą krawędź}, to wystarczy sprawdzić $|Q|$ opcji.
    
    Jeśli ma \textbf{$k$ brakujących krawędzi}, to wystarczy sprawdzić $|Q|^k$ opcji.
    
    \textbf{Pytanie:} Czy możemy rozdzielić trudność problemu od rozmiaru automatu?
  \end{block}
  
  \vspace{0.2cm}
  
  \begin{block}{Złożoność parametryczna}
    Framework analizy, gdzie oprócz rozmiaru wejścia ($n$) mamy parametr $k$.
    
    \textbf{Cel:} Znaleźć algorytm działający w czasie $f(k) \cdot n^c$ zamiast $n^{k+c}$ (co należy do klasy XP).
  \end{block}
\end{frame}

\begin{frame}{$W[2]$-trudność: redukcja z $k$-set cover}
  \begin{block}{Problem $k$-set cover (znany $W[2]$-zupełny)}
    \small
    \textbf{Wejście:} Uniwersum $U$, rodzina zbiorów $\mathcal{F}$, liczba $k$
    
    \textbf{Wyjście:} Czy $\leq k$ zbiorów z $\mathcal{F}$ pokrywa całe $U$?
  \end{block}
  
  \vspace{0.2cm}
  
  \begin{block}{Nasza redukcja}
    Dla każdego zbioru $S_i \in \mathcal{F}$ tworzymy stan $s_i$ w automacie.
    
    Dla każdego elementu $u_j \in U$ tworzymy literę $e_j$.
    
    Brakujące krawędzie $p_1, \ldots, p_k$ reprezentują wybór zbiorów do pokrycia.
  \end{block}
  
  \vspace{0.2cm}
  
  \begin{block}{Kluczowe próbki}
    \small
    \begin{itemize}
      \item $p_1 t e_j p_2 t e_j \cdots p_k t e_j \in S^+$ dla każdego $u_j$
      
      (gwarantuje że każdy element jest pokryty przez wybrany zbiór)
      
      \item $p_x \in S^-$ dla każdego $x \in \{1, \ldots, k\}$
      
      (wymusza że każde $p_x$ prowadzi do stanu $s_i$, nie do stanu akceptującego)
    \end{itemize}
  \end{block}
\end{frame}

\begin{frame}{$W[2]$-trudność: rozmiar alfabetu}
  \begin{block}{Problem z poprzedniej redukcji}
    Alfabet ma rozmiar zależny od $|U|$ — to słaby punkt redukcji.
  \end{block}
  
  \vspace{0.2cm}
  
  \begin{block}{Ulepszona redukcja: alfabet binarny}
    Można pokazać redukcję do problemu z alfabetem $\Sigma = \{0, 1\}$.
    
    Idea: zamiast jednej litery $e_j$ dla każdego elementu, używamy bin. reprezentacji indeksu.
    
    \textbf{Wynik:} Problem jest $W[2]$-trudny nawet z alfabetem $|\Sigma| = 2$.
  \end{block}
\end{frame}

\begin{frame}{Przynależność do $W[P]$}
  \begin{block}{Algorytm niedeterministyczny}
    \small
    Niedeterministyczna maszyna Turinga:
    \begin{enumerate}
      \item \textbf{Zgaduj:} $O(k \log |Q|)$ bitów niedeterminizmu
      
      \mbox{Zagadnij $k$ stanów docelowych brakujących krawędzi}
      
      \item \textbf{Weryfikuj:} Czas wielomianowy
      
      \mbox{Sprawdź czy automat akceptuje $S^+$ i odrzuca $S^-$}
    \end{enumerate}
  \end{block}
  
  \vspace{0.2cm}
  
  \begin{block}{Wniosek}
    Problem należy do $W[P]$ — mamy $\kappa$-ograniczoną niedeterministyczną maszynę Turinga.
  \end{block}
\end{frame}

\begin{frame}{Rezultat: złożoność naszego problemu}
  \begin{block}{Twierdzenie}
    Problem naprawienia $k$-częściowego DFA jest $W[2]$-trudny i należy do $W[P]$.
    
    $$W[2] \subseteq \{\text{nasz problem}\} \subseteq W[P]$$
  \end{block}
  
  \vspace{0.3cm}
  
  \begin{block}{Co to oznacza?}
    \begin{itemize}
      \item \textbf{Problem nie jest FPT} (chyba że FPT $= W[2]$, co jest problemem milenialnym)
      \item \textbf{Problem nie jest XP} (bo wtedy byłby przynajmniej w $W[P]$ bez górnego ograniczenia)
      \item \textbf{Problem jest trudny parametrycznie} — trudniej niż cokolwiek w FPT
      \item \textbf{Ale mamy certyfikat w $W[P]$} — nie jest całkowicie bez nadziei
    \end{itemize}
  \end{block}
\end{frame}
